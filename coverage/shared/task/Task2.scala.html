8 /**
9  * Task with effect tracking.
10  *
11  * @tparam R The required capabilities (environment) - what this task needs to run
12  * @tparam A The success type
13  *
14  * Example:
15  *   def getUser(id: String): Task2[Database, User]
16  *   def log(msg: String): Task2[Logging, Unit]
17  *   def getUserAndLog(id: String): Task2[Database & Logging, User]
18  */
19 opaque type Task2[-R, +A] = R => Result[A]
20 
21 object Task2:
22   // === Constructors ===
23 
24   /** Create a task that requires capability R */
25   def apply[R, A](f: R => Result[A]): Task2[R, A] = f
26 
27   /** Create a task from a result (no requirements) */
28   def fromResult[A](result: Result[A]): Task2[Any, A] = _ => result
29 
30   /** Wrap a pure value (no requirements, cannot fail) */
31   def succeed[A](a: A): Task2[Any, A] = _ => Right(a)
32 
33   /** Create a failed task */
34   def fail[A](context: String, cause: Any): Task2[Any, A] = _ => Left(Fail(context, cause))
35 
36   /** Access a capability from the environment */
37   def service[R]: Task2[R, R] = r => Right(r)
38 
39   /** Create a task that uses a specific service */
40   def serviceWith[R, A](f: R => A): Task2[R, A] = r => Right(f(r))
41 
42   /** Create a task that uses a service and may fail */
43   def serviceWithTask[R, A](f: R => Result[A]): Task2[R, A] = f
44 
45   /** Lift a Try into a Task2 */
46   def fromTry[A](context: String)(t: => Try[A]): Task2[Any, A] = _ => t.toResult(context)
47 
48   /** Lift an Option into a Task2 */
49   def fromOption[A](context: String, ifNone: => Any = "not found")(o: => Option[A]): Task2[Any, A] =
50     _ => o.toResult(context, ifNone)
51 
52   /** Lift an Either into a Task2 */
53   def fromEither[A](context: String)(e: => Either[Any, A]): Task2[Any, A] =
54     _ => e.left.map(err => Fail(context, err))

56   // === Extension methods ===
57 
58   extension [R, A](self: Task2[R, A])
59     /** Run the task with the provided environment */
60     def run(env: R): Result[A] = self(env)
61 
62     /** Run a task that has no requirements */
63     def runUnit(using ev: Any <:< R): Result[A] = self(())
64 
65     def map[B](f: A => B): Task2[R, B] =
66       r => self(r).map(f)
67 
68     def flatMap[R1 <: R, B](f: A => Task2[R1, B]): Task2[R1, B] =
69       r => self(r) match
70         case Left(fail) => Left(fail)
71         case Right(a) => f(a)(r)
72 
73     /** Combine with another task, running them sequentially */
74     def zip[R1 <: R, B](other: Task2[R1, B]): Task2[R1, (A, B)] =
75       r => for
76         a <- self(r)
77         b <- other(r)
78       yield (a, b)
79 
80     /** Combine with another task, running them in parallel */
81     def zipPar[R1 <: R, B](other: Task2[R1, B])(using ExecutionContext): Task2[R1, (A, B)] =
82       r =>
83         val futureA = Future(self(r))
84         val futureB = Future(other(r))
85         val combined = for
86           a <- futureA
87           b <- futureB
88         yield (a, b) match
89           case (Right(av), Right(bv)) => Right((av, bv))
90           case (Left(f), _) => Left(f)
91           case (_, Left(f)) => Left(f)
92         Await.result(combined, 30.seconds)
93 
94     def retry(attempts: Int): Task2[R, A] =
95       r =>
96         @tailrec
97         def loop(remaining: Int): Result[A] =
98           self(r) match
99             case Right(a) => Right(a)
100             case Left(f) if remaining > 1 => loop(remaining - 1)
101             case Left(f) => Left(f)
102         loop(attempts)
103 
104     def timeout(duration: FiniteDuration)(using ExecutionContext): Task2[R, A] =
105       r =>
106         val future = Future(self(r))
107         Try(Await.result(future, duration)) match
108           case scala.util.Success(result) => result
109           case scala.util.Failure(_: java.util.concurrent.TimeoutException) =>
110             Left(Fail("timeout", s"timed out after $duration"))
111           case scala.util.Failure(err) =>
112             Left(Fail("timeout", err.getMessage))
113 
114     /** Provide part of the environment */
115     def provide[R0](layer: R0 => R): Task2[R0, A] =
116       r0 => self(layer(r0))
117 
118     /** Provide the entire environment, eliminating the requirement */
119     def provideEnvironment(env: R): Task2[Any, A] =
120       _ => self(env)
121 
122     /** Recover from failures */
123     def recover(f: Fail => Result[A]): Task2[R, A] =
124       r => self(r) match
125         case Left(fail) => f(fail)
126         case right => right
127 
128     /** Ensure cleanup runs regardless of success/failure */
129     def ensuring(cleanup: => Unit): Task2[R, A] =
130       r =>
131         val result = Try(self(r)) match
132           case scala.util.Success(res) => res
133           case scala.util.Failure(ex) =>
134             cleanup
135             Left(Fail("ensuring", ex.getMessage))
136         cleanup
137         result
