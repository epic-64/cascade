96 // Player inventory
97 case class Inventory(
98     items: Map[Item, Int] // Item -> quantity
99 ) derives ReadWriter:
100   def totalWeight: Int =
101     items.map((item, qty) => Item.weight(item) * qty).sum
102 
103   def add(item: Item, qty: Int): Inventory =
104     copy(items = items.updated(item, items.getOrElse(item, 0) + qty))
105 
106   def remove(item: Item, qty: Int): Option[Inventory] =
107     val current = items.getOrElse(item, 0)
108     if current >= qty then
109       val newQty = current - qty
110       val newItems = if newQty == 0 then items - item else items.updated(item, newQty)
111       Some(copy(items = newItems))
112     else None
113 
114   def getQuantity(item: Item): Int = items.getOrElse(item, 0)
