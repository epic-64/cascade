5 /** Risk system for Trader game.
6   *
7   * High-value cargo attracts bandits. The risk scales with value per kg:
8   *   - Gems (40 g/kg) = very high risk
9   *   - Bulk goods (<1 g/kg) = essentially safe
10   *   - Mixed cargo = proportionally reduced risk
11   */
12 object TraderRisk:
13 
14   // Risk formula constants (can be tuned for balance)
15   private val BaseDivisor: Double = 10.0 // Higher = lower risk overall
16   private val RiskExponent: Double = 1.5 // Higher = steeper penalty for high-value cargo
17   private val EncounterMultiplier: Double = 0.1 // Higher = more frequent encounters
18   private val EncounterCap: Double = 0.8 // Maximum encounter chance (80%)
19 
20   /** Calculate the risk assessment for current cargo. Uses item base prices (not market prices) for consistent risk
21     * calculation.
22     */
23   def assessRisk(inventory: Inventory): RiskAssessment =
24     val totalWeight = inventory.totalWeight
25 
26     if totalWeight == 0 then
27       RiskAssessment.safe
28     else
29       val totalValue = inventory.items.map { (item, qty) =>
30         Item.basePrice(item) * qty
31       }.sum
32 
33       val valuePerKg = totalValue.toDouble / totalWeight
34       val riskScore = math.pow(valuePerKg / BaseDivisor, RiskExponent)
35       val encounterChance = math.min(riskScore * EncounterMultiplier, EncounterCap)
36 
37       RiskAssessment(totalValue, totalWeight, valuePerKg, riskScore, encounterChance)
38 
39   /** Roll for a bandit encounter during travel. Returns None if no encounter, Some(outcome) if bandits strike.
40     */
41   def rollEncounter(risk: RiskAssessment, rng: Random): Option[EncounterOutcome] =
42     if risk.encounterChance <= 0.0 then None
43     else if rng.nextDouble() >= risk.encounterChance then None
44     else Some(resolveEncounter(risk, rng))

46   /** Determine the outcome of a bandit encounter.
47     *
48     * Probability distribution:
49     *   - 20% Escaped (no loss)
50     *   - 40% Toll (10-20% cargo value in gold)
51     *   - 25% Robbery (30-50% of highest-value items)
52     *   - 15% Devastating Loss (50-80% of all cargo)
53     */
54   private def resolveEncounter(risk: RiskAssessment, rng: Random): EncounterOutcome =
55     val roll = rng.nextInt(100)
56     if roll < 20 then
57       EncounterOutcome.Escaped
58     else if roll < 60 then
59       val lossPercent = 0.10 + rng.nextDouble() * 0.10 // 10-20%
60       val goldLost = math.max(1, (risk.cargoValue * lossPercent).toInt)
61       EncounterOutcome.Toll(goldLost)
62     else if roll < 85 then
63       EncounterOutcome.Robbery(Map.empty) // Placeholder - calculated in applyEncounter
64     else
65       EncounterOutcome.DevastatingLoss(Map.empty, 0) // Placeholder - calculated in applyEncounter

67   /** Apply encounter outcome to game state. Returns updated game with losses applied and log entry added.
68     */
69   def applyEncounter(
70       game: TraderGame,
71       outcome: EncounterOutcome,
72       fromCity: CityId,
73       toCity: CityId,
74       rng: Random
75   ): TraderGame =
76     val turn = game.turn
77 
78     outcome match
79       case EncounterOutcome.Unscathed =>
80         // Safe arrival - this case is handled in TraderLogic.travel directly
81         // but we include it here for completeness
82         val encounter = BanditEncounter(EncounterOutcome.Unscathed, fromCity, toCity, turn)
83         game.copy(lastEncounter = Some(encounter))
84 
85       case EncounterOutcome.Escaped =>
86         val encounter = BanditEncounter(EncounterOutcome.Escaped, fromCity, toCity, turn)
87         val logEntry = "⚔️ Bandits attacked but you escaped unharmed!"
88         game.copy(
89           lastEncounter = Some(encounter),
90           log = (logEntry :: game.log).take(TraderGame.MaxLogEntries)
91         )
92 
93       case EncounterOutcome.Toll(goldLost) =>
94         val actualLoss = math.min(goldLost, game.player.gold)
95         val actualOutcome = EncounterOutcome.Toll(actualLoss)
96         val encounter = BanditEncounter(actualOutcome, fromCity, toCity, turn)
97         val newPlayer = game.player.copy(gold = game.player.gold - actualLoss)
98         val logEntry = s"⚔️ Bandits demanded a toll! Lost ${actualLoss}g"
99         game.copy(
100           player = newPlayer,
101           lastEncounter = Some(encounter),
102           log = (logEntry :: game.log).take(TraderGame.MaxLogEntries)
103         )
104 
105       case EncounterOutcome.Robbery(_) =>
106         // Calculate actual robbery: 30-50% of highest-value items
107         val lossPercent = 0.30 + rng.nextDouble() * 0.20
108         val itemsLost = calculateRobbery(game.player.inventory, lossPercent, rng)
109         val actualOutcome = EncounterOutcome.Robbery(itemsLost)
110         val encounter = BanditEncounter(actualOutcome, fromCity, toCity, turn)
111         val newInventory = removeItems(game.player.inventory, itemsLost)
112         val newPlayer = game.player.copy(inventory = newInventory)
113         val lostItemsDesc = formatItemsLost(itemsLost)
114         val logEntry = s"⚔️ Bandits robbed your cargo! Lost $lostItemsDesc"
115         game.copy(
116           player = newPlayer,
117           lastEncounter = Some(encounter),
118           log = (logEntry :: game.log).take(TraderGame.MaxLogEntries)
119         )
120 
121       case EncounterOutcome.DevastatingLoss(_, _) =>
122         // Calculate devastating loss: 50-80% of all cargo + some gold
123         val lossPercent = 0.50 + rng.nextDouble() * 0.30
124         val itemsLost = calculateDevastatingLoss(game.player.inventory, lossPercent, rng)
125         val goldLost = math.min((game.player.gold * 0.2).toInt, game.player.gold)
126         val actualOutcome = EncounterOutcome.DevastatingLoss(itemsLost, goldLost)
127         val encounter = BanditEncounter(actualOutcome, fromCity, toCity, turn)
128         val newInventory = removeItems(game.player.inventory, itemsLost)
129         val newPlayer = game.player.copy(
130           gold = game.player.gold - goldLost,
131           inventory = newInventory
132         )
133         val lostItemsDesc = formatItemsLost(itemsLost)
134         val goldDesc = if goldLost > 0 then s" and ${goldLost}g" else ""
135         val logEntry = s"⚔️ Devastating bandit attack! Lost $lostItemsDesc$goldDesc"
136         game.copy(
137           player = newPlayer,
138           lastEncounter = Some(encounter),
139           log = (logEntry :: game.log).take(TraderGame.MaxLogEntries)
140         )

142   /** Calculate which items are lost in a robbery. Targets highest value-per-kg items first.
143     */
144   private def calculateRobbery(
145       inventory: Inventory,
146       lossPercent: Double,
147       rng: Random
148   ): Map[Item, Int] =
149     val itemsByValue = inventory.items.toList
150       .sortBy((item, _) => -Item.basePrice(item).toDouble / Item.weight(item))
151 
152     val totalValue = inventory.items.map((item, qty) => Item.basePrice(item) * qty).sum
153     val targetLossValue = (totalValue * lossPercent).toInt
154 
155     var lostValue = 0
156     var result = Map.empty[Item, Int]
157 
158     itemsByValue.foreach { (item, qty) =>
159       if lostValue < targetLossValue && qty > 0 then
160         val itemValue = Item.basePrice(item)
161         val maxToLose = math.ceil((targetLossValue - lostValue).toDouble / itemValue).toInt
162         val qtyToLose = math.min(maxToLose, qty)
163         if qtyToLose > 0 then
164           result = result.updated(item, qtyToLose)
165           lostValue += qtyToLose * itemValue
166     }
167 
168     result

170   /** Calculate which items are lost in a devastating attack. Loses a percentage of ALL items.
171     */
172   private def calculateDevastatingLoss(
173       inventory: Inventory,
174       lossPercent: Double,
175       rng: Random
176   ): Map[Item, Int] =
177     inventory.items.map { (item, qty) =>
178       val qtyToLose = math.max(1, (qty * lossPercent).toInt)
179       item -> math.min(qtyToLose, qty)
180     }.filter((_, qty) => qty > 0)
181 
182   /** Remove items from inventory */
183   private def removeItems(inventory: Inventory, itemsToRemove: Map[Item, Int]): Inventory =
184     itemsToRemove.foldLeft(inventory) { case (inv, (item, qty)) =>
185       inv.remove(item, qty).getOrElse(inv)
186     }

188   /** Format lost items for log display */
189   private def formatItemsLost(items: Map[Item, Int]): String =
190     if items.isEmpty then "nothing"
191     else items.map((item, qty) => s"$qty ${item.toString}").mkString(", ")
