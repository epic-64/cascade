7   // Cities are on a 3x3 grid, distance calculated as Manhattan distance with diagonals = 1.5
8   private def distance(from: City, to: City): Double =
9     val (r1, c1) = from.position
10     val (r2, c2) = to.position
11     val rowDiff = math.abs(r1 - r2)
12     val colDiff = math.abs(c1 - c2)
13     // Each step is 1 unit, diagonals count as 1.5
14     val diagonals = math.min(rowDiff, colDiff)
15     val straights = math.max(rowDiff, colDiff) - diagonals
16     straights + diagonals * 1.5

18   def travelCost(from: City, to: City, cargoWeight: Int): Int =
19     val dist = distance(from, to)
20     val baseCost = dist * 5
21     val cargoCost = cargoWeight * 0.02
22     math.ceil(baseCost + cargoCost).toInt

24   /** Calculate the market price for an item in a city. This is the unified price - what you pay to buy AND what you
25     * receive when selling. Price is affected by:
26     *   - Base item price
27     *   - City's supply level (abundant = cheaper, scarce = expensive)
28     *   - City's demand level (high = expensive, low = cheaper)
29     *   - Current season modifiers
30     */
31   def calculatePrice(city: City, item: Item, season: Season): Int =
32     val base = Item.basePrice(item)
33     val supplyMod = city.market.supply.get(item).map(SupplyLevel.modifier).getOrElse(1.0)
34     val demandMod = city.market.demand.get(item).map(DemandLevel.modifier).getOrElse(1.0)
35     val seasonMod = Season.modifier(season, item)
36     math.ceil(base * supplyMod * demandMod * seasonMod).toInt
37

45   def buyItem(game: TraderGame, item: Item, qty: Int): Either[String, TraderGame] =
46     if qty <= 0 then return Left("Quantity must be positive")
47 
48     val price = calculateBuyPrice(game.currentCity, item, game.season) * qty
49     val weight = Item.weight(item) * qty
50 
51     if price > game.player.gold then
52       Left(s"Not enough gold. Need $price, have ${game.player.gold}")
53     else if weight > game.player.availableCapacity then
54       Left(s"Not enough capacity. Need ${weight}kg, have ${game.player.availableCapacity}kg")
55     else
56       val newPlayer = game.player.copy(
57         gold = game.player.gold - price,
58         inventory = game.player.inventory.add(item, qty)
59       )
60       val logEntry = s"Bought $qty ${item.toString} for ${price}g"
61       Right(game.copy(
62         player = newPlayer,
63         log = (logEntry :: game.log).take(TraderGame.MaxLogEntries)
64       ))

66   def sellItem(game: TraderGame, item: Item, qty: Int): Either[String, TraderGame] =
67     if qty <= 0 then return Left("Quantity must be positive")
68 
69     val currentQty = game.player.inventory.getQuantity(item)
70     if currentQty < qty then
71       Left(s"Not enough ${item.toString}. Have $currentQty, trying to sell $qty")
72     else
73       val price = calculateSellPrice(game.currentCity, item, game.season) * qty
74       game.player.inventory.remove(item, qty) match
75         case Some(newInventory) =>
76           val newPlayer = game.player.copy(
77             gold = game.player.gold + price,
78             inventory = newInventory
79           )
80           val logEntry = s"Sold $qty ${item.toString} for ${price}g"
81           Right(game.copy(
82             player = newPlayer,
83             log = (logEntry :: game.log).take(TraderGame.MaxLogEntries)
84           ))
85         case None =>
86           Left(s"Failed to remove items from inventory")

92   def travel(game: TraderGame, destination: CityId, rng: Random): Either[String, TraderGame] =
93     if destination == game.player.currentCity then
94       Left("Already in this city")
95     else
96       val fromCity = game.currentCity
97       val toCity = game.cities(destination)
98       val cost = travelCost(fromCity, toCity, game.player.inventory.totalWeight)
99 
100       if cost > game.player.gold then
101         Left(s"Not enough gold for travel. Need ${cost}g, have ${game.player.gold}g")
102       else
103         val newPlayer = game.player.copy(
104           gold = game.player.gold - cost,
105           currentCity = destination
106         )
107         val logEntry = s"Traveled to ${toCity.name} (-${cost}g)"
108         val gameAfterTravel = game.copy(
109           player = newPlayer,
110           visitedCities = game.visitedCities + destination,
111           log = (logEntry :: game.log).take(TraderGame.MaxLogEntries)
112         )
113 
114         // Check for bandit encounter - always set lastEncounter for UI feedback
115         val risk = TraderRisk.assessRisk(game.player.inventory)
116         val gameAfterEncounter = TraderRisk.rollEncounter(risk, rng) match
117           case Some(outcome) =>
118             TraderRisk.applyEncounter(gameAfterTravel, outcome, game.player.currentCity, destination, rng)
119           case None =>
120             // Safe arrival - record Unscathed outcome
121             val encounter = BanditEncounter(EncounterOutcome.Unscathed, game.player.currentCity, destination, game.turn)
122             gameAfterTravel.copy(lastEncounter = Some(encounter))
123 
124         Right(advanceTurn(gameAfterEncounter))

145   private def advanceTurn(game: TraderGame): TraderGame =
146     val newTurn = game.turn + 1
147     // Season changes every 5 turns
148     if newTurn % 5 == 1 && newTurn > 1 then
149       changeSeason(game.copy(turn = newTurn))

153   private def changeSeason(game: TraderGame): TraderGame =
154     val newSeason = Season.next(game.season)
155     val logEntry = s"Season changed to ${newSeason.toString}! Markets have shifted."
156     // Shuffle market conditions for all cities

163       visitedCities = Set(game.player.currentCity), // Reset - only current city is known
164       log = (logEntry :: game.log).take(TraderGame.MaxLogEntries)

167   private def shuffleMarket(city: City, cityId: CityId): CityMarket =
168     val (highSupply, highDemand) = citySpecialization(cityId)
169 
170     val supply = Item.all.map { item =>
171       val level =
172         if highSupply.contains(item) then SupplyLevel.Abundant
173         else randomSupplyLevel()
174       item -> level
175     }.toMap
176 
177     val demand = Item.all.map { item =>
178       val level =
179         if highDemand.contains(item) then DemandLevel.High
180         else randomDemandLevel()
181       item -> level
182     }.toMap
183 
184     CityMarket(supply, demand)

186   private def randomSupplyLevel(): SupplyLevel =
187     val rand = Random.nextDouble()
