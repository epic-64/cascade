5 /** Offline progression calculator.
6   *
7   * When the player returns after being away, this calculates what they would
8   * have gained during that time and applies it in bulk.
9   *
10   * == Accuracy Notes ==
11   * - Gathering/Processing/Thieving: XP gains are exact (actions × XP per action)
12   * - Combat: Simulated tick-by-tick using real game logic (100% accurate)
13   * - Rare drops use expected value for non-combat (may vary slightly from real gameplay)
14   * - Efficiency/yield bonuses are calculated once at start for non-combat
15   * - Potion and tablet effects are NOT applied (they'd expire quickly anyway)
16   * - Inventory overflow is ignored (items may exceed normal limits)
17   *
18   * == Maintenance Requirements ==
19   * When modifying gameplay systems, update this file if you:
20   * - Add a new ActiveAction type (add a new case in calculateOfflineProgress)
21   * - Change XP formulas in SkillState/ActionState (formulas are referenced directly)
22   * - Add new bonus calculations to VelorIdleLogic (may need to include here)
23   * - Change how processing/gathering/thieving works (update respective process* methods)
24   * - Combat changes are automatically picked up (uses VelorIdleLogic.tick directly)
25   *
26   * == Not Supported ==
27   * - Astrology (not implemented yet)
28   */
29 object OfflineProgress:

52   /** Calculate and apply offline progression.
53     *
54     * @param game Current game state
55     * @param lastTickTime When the game was last updated
56     * @param currentTime Current timestamp
57     * @param random Random source for drops
58     * @return Updated game state and summary of gains
59     */
60   def calculateOfflineProgress(
61     game: VelorIdleGame,
62     lastTickTime: Long,
63     currentTime: Long,
64     random: Random = Random
65   ): OfflineResult =
66     val elapsedSeconds = ((currentTime - lastTickTime) / 1000L).min(MaxOfflineSeconds)
67 
68     if elapsedSeconds < ChunkDurationSeconds then
69       // Not enough time passed for offline processing
70       return OfflineResult(game, 0, 0, Map.empty, Map.empty, Map.empty, Map.empty, 0, Map.empty, Map.empty, Vector.empty)
71 
72     game.activeAction match
73       case ActiveAction.Idle =>
74         OfflineResult(game, elapsedSeconds, 0, Map.empty, Map.empty, Map.empty, Map.empty, 0, Map.empty, Map.empty, Vector.empty)
75 
76       case ActiveAction.Gathering(skill, action) =>
77         processGatheringOffline(game, skill, action, elapsedSeconds, random)
78 
79       case ActiveAction.Processing(skill, action) =>
80         processProcessingOffline(game, skill, action, elapsedSeconds, random)
81 
82       case ActiveAction.Thieving(action) =>
83         processThievingOffline(game, action, elapsedSeconds, random)
84 
85       case ActiveAction.EquipmentCrafting(action) =>
86         processEquipmentCraftingOffline(game, action, elapsedSeconds, random)
87 
88       case ActiveAction.Adventure =>
89         processAdventureOffline(game, lastTickTime, currentTime, elapsedSeconds, random)
90 
91       case ActiveAction.Rest =>
92         processRestOffline(game, elapsedSeconds)

98   private def processGatheringOffline(
99     game: VelorIdleGame,
100     skill: Skill,
101     action: GatheringAction,
102     elapsedSeconds: Long,
103     random: Random
104   ): OfflineResult =
105     val actionState = game.actionLevels.getOrElse(action.id, ActionState.initial)
106     val skillState = game.skills.getOrElse(skill, SkillState.initial)
107 
108     // Calculate effective time per action (with efficiency bonus)
109     val efficiency = VelorIdleLogic.calculateEfficiencyBonus(actionState.level)
110     val effectiveTime = action.timeSeconds * (1.0 - efficiency)
111 
112     // Calculate number of actions that would complete
113     val actionsCompleted = (elapsedSeconds / effectiveTime).toLong
114     val chunksProcessed = (elapsedSeconds / ChunkDurationSeconds).toInt
115 
116     if actionsCompleted <= 0 then
117       return OfflineResult(game, elapsedSeconds, 0, Map.empty, Map.empty, Map.empty, Map.empty, 0, Map.empty, Map.empty, Vector.empty)
118 
119     // Calculate XP gained
120     val skillXpGained = actionsCompleted * action.xpGain
121     val actionXpGained = actionsCompleted * action.xpGain
122 
123     // Calculate items gained (base + yield bonus)
124     val yieldBonus = VelorIdleLogic.calculateYieldBonus(actionState.level)
125     val avgItemsPerAction = 1.0 + yieldBonus
126     val baseItems = (actionsCompleted * avgItemsPerAction).toLong
127 
128     // Roll for rare drops
129     val rareItems = action.rareOutput match
130       case Some((item, chance)) =>
131         val expectedRare = (actionsCompleted * chance).toLong
132         val extraRare = if random.nextDouble() < (actionsCompleted * chance - expectedRare) then 1L else 0L
133         Map(item -> (expectedRare + extraRare)).filter(_._2 > 0)
134       case None => Map.empty[Item, Long]
135 
136     val itemsGained = Map(action.output -> baseItems) ++ rareItems
137 
138     // Apply XP and calculate level ups
139     var updatedGame = game
140     var events = Vector.empty[GameEvent]
141 
142     // Apply skill XP
143     val oldSkillLevel = skillState.level
144     val newSkillXp = skillState.xp + skillXpGained
145     val newSkillLevel = SkillState.levelFromXp(newSkillXp)
146     val newSkillState = skillState.copy(xp = newSkillXp, level = newSkillLevel)
147     updatedGame = updatedGame.copy(skills = updatedGame.skills.updated(skill, newSkillState))
148 
149     val skillLevelUps = if newSkillLevel > oldSkillLevel then
150       events :+= GameEvent.LevelUp(skill, newSkillLevel)
151       Map(skill -> (newSkillLevel - oldSkillLevel))
152     else Map.empty[Skill, Int]
153 
154     // Apply action XP
155     val oldActionLevel = actionState.level
156     val newActionXp = actionState.xp + actionXpGained
157     val newActionLevel = ActionState.levelFromXp(newActionXp, action.levelRequired)
158     val newActionState = actionState.copy(xp = newActionXp, level = newActionLevel)
159     updatedGame = updatedGame.copy(actionLevels = updatedGame.actionLevels.updated(action.id, newActionState))
160 
161     val actionLevelUps = if newActionLevel > oldActionLevel then
162       events :+= GameEvent.ActionLevelUp(action.id, newActionLevel)
163       Map(action.id -> (newActionLevel - oldActionLevel))
164     else Map.empty[String, Int]
165 
166     // Add items to inventory
167     itemsGained.foreach { case (item, count) =>
168       val (newInv, _) = updatedGame.inventory.addItem(item, count)
169       updatedGame = updatedGame.copy(inventory = newInv)
170     }
171 
172     // Award combat skill points if Adventure leveled up
173     if skill == Skill.Adventure && newSkillLevel > oldSkillLevel then
174       val pointsGained = newSkillLevel - oldSkillLevel
175       val newCombatSkillState = SkillTreeLogic.awardPoints(updatedGame.adventureState.combatSkillState, pointsGained)
176       updatedGame = updatedGame.copy(
177         adventureState = updatedGame.adventureState.copy(combatSkillState = newCombatSkillState)
178       )
179 
180     OfflineResult(
181       game = updatedGame,
182       secondsProcessed = elapsedSeconds,
183       chunksProcessed = chunksProcessed,
184       xpGained = Map(skill -> skillXpGained),
185       actionXpGained = Map(action.id -> actionXpGained),
186       itemsGained = itemsGained,
187       itemsConsumed = Map.empty, // Gathering doesn't consume items
188       goldGained = 0,
189       skillLevelUps = skillLevelUps,
190       actionLevelUps = actionLevelUps,
191       events = events
192     )

503   /** Tick interval for offline combat simulation in milliseconds */
504   private val CombatTickIntervalMs: Long = 100L
505   
506   /** Maximum number of combat ticks to simulate (prevents infinite loops) */
507   private val MaxCombatTicks: Int = 100_000 // ~2.7 hours at 100ms ticks
508 
509   private def processAdventureOffline(
510     game: VelorIdleGame,
511     lastTickTime: Long,
512     currentTime: Long,
513     elapsedSeconds: Long,
514     random: Random
515   ): OfflineResult =
516     val chunksProcessed = (elapsedSeconds / ChunkDurationSeconds).toInt
517     
518     // Track gains across all combat
519     var totalXpGained = 0L
520     var totalGoldGained = 0L
521     var itemsGained = Map.empty[Item, Long]
522     var allEvents = Vector.empty[GameEvent]
523     var enemiesDefeated = 0
524     
525     // Simulate combat tick by tick
526     var currentGame = game
527     var simulatedTime = lastTickTime
528     var tickCount = 0
529     
530     while simulatedTime < currentTime && tickCount < MaxCombatTicks do
531       val tickEndTime = (simulatedTime + CombatTickIntervalMs).min(currentTime)
532       
533       val (updatedGame, events) = VelorIdleLogic.tick(currentGame, tickEndTime, random)
534       
535       // Track XP gains
536       events.foreach:
537         case GameEvent.XpGained(skill, amount) if skill == Skill.Adventure =>
538           totalXpGained += amount
539         case GameEvent.GoldGained(amount) =>
540           totalGoldGained += amount
541         case GameEvent.ItemGained(item, count) =>
542           itemsGained = itemsGained.updated(item, itemsGained.getOrElse(item, 0L) + count)
543         case GameEvent.AdventureEnemyDefeated(_) =>
544           enemiesDefeated += 1
545         case _ => ()
546       
547       allEvents ++= events
548       currentGame = updatedGame
549       simulatedTime = tickEndTime
550       tickCount += 1
551       
552       // If combat ended (player died or stopped), break out
553       if currentGame.activeAction != ActiveAction.Adventure then
554         // If player died and is now resting, let them fully recover
555         if currentGame.activeAction == ActiveAction.Rest then
556           val advState = currentGame.adventureState
557           currentGame = currentGame.copy(
558             adventureState = advState.copy(
559               currentHp = advState.maxHp,
560               currentMana = advState.maxMana
561             ),
562             activeAction = ActiveAction.Idle
563           )
564         simulatedTime = currentTime // Exit the loop
565     
566     // Calculate level ups
567     val oldLevel = game.skills.getOrElse(Skill.Adventure, SkillState.initial).level
568     val newLevel = currentGame.skills.getOrElse(Skill.Adventure, SkillState.initial).level
569     val skillLevelUps = if newLevel > oldLevel then 
570       Map(Skill.Adventure -> (newLevel - oldLevel)) 
571     else Map.empty[Skill, Int]
572     
573     OfflineResult(
574       game = currentGame.copy(lastTickTime = currentTime),
575       secondsProcessed = elapsedSeconds,
576       chunksProcessed = chunksProcessed,
577       xpGained = if totalXpGained > 0 then Map(Skill.Adventure -> totalXpGained) else Map.empty,
578       actionXpGained = Map.empty,
579       itemsGained = itemsGained,
580       itemsConsumed = Map.empty, // Combat doesn't consume items
