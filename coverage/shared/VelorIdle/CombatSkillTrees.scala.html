18 /** A skill within a skill tree */
19 case class TreeSkill(
20   id: String,
21   name: String,
22   icon: String,
23   description: String,
24   manaCost: Int,                 // Base mana cost at level 1
25   cooldownMs: Long,
26   baseDamage: Int,               // Base damage at level 1
27   damagePerLevel: Int,           // Additional damage per level
28   castTimeMs: Long = 0L,
29   effects: Vector[SkillEffect] = Vector.empty,
30   chainSkills: Vector[TreeChainSkill] = Vector.empty,  // Built-in chain skills
31   maxLevel: Int = 10,            // Maximum level for this skill
32   unlockRequirement: Option[TreeSkillRequirement] = None,  // Required skill level to unlock
33   isSpell: Boolean = false       // True for spells (higher mana scaling), false for melee attacks
34 ) derives ReadWriter:
35   /** Calculate damage at a given level using triangular growth.
36     * Level 1: baseDamage
37     * Level 2: baseDamage + 1 * damagePerLevel (triangular(1) = 1)
38     * Level 3: baseDamage + 3 * damagePerLevel (triangular(2) = 3)
39     * Level 4: baseDamage + 6 * damagePerLevel (triangular(3) = 6)
40     * etc.
41     */
42   def damageAtLevel(level: Int): Int =
43     if level <= 0 then 0
44     else baseDamage + (AdventureState.triangular(level - 1) * damagePerLevel)
45 
46   /** Calculate mana cost at a given level.
47     * Spells: +0.5 mana per level
48     * Attacks: +0.5 mana per 5 levels (i.e., +0.1 per level)
49     */
50   def manaCostAtLevel(level: Int): Int =
51     if level <= 1 then manaCost
52     else
53       val extraLevels = level - 1
54       val extraMana = if isSpell then extraLevels * 0.5 else extraLevels * 0.1
55       (manaCost + extraMana).toInt

628   /** Get the first unlocked chain skill (if any) at the given level */
629   private def getUnlockedChainSkill(treeSkill: TreeSkill, level: Int): Option[ChainSkill] =
630     treeSkill.chainSkills.headOption.flatMap { chain =>
631       if level >= chain.requiredLevel then
632         Some(ChainSkill(
633           skill = toCombatSkill(chain.skill, level),  // Recursive - will check nested chain requirements
634           windowMs = chain.windowMs

640   /** Build combat skill slots from bound skills and their levels */
641   def buildSkillSlots(combatSkillState: CombatSkillState, equipmentSkillBonuses: Map[String, Int] = Map.empty): Vector[SkillSlotState] =
642     combatSkillState.boundSkills.map { maybeSkillId =>
643       maybeSkillId.flatMap(SkillTrees.getSkillById) match
644         case Some(treeSkill) =>
645           val baseLevel = combatSkillState.getSkillLevel(treeSkill.id).max(1)
646           // Find which tree this skill belongs to for equipment bonuses
647           val treeId = SkillTrees.all.find(_.skills.exists(_.id == treeSkill.id)).map(_.id).getOrElse("")
648           val equipBonus = equipmentSkillBonuses.getOrElse(treeId, 0)
649           val effectiveLevel = baseLevel + equipBonus
650           val combatSkill = toCombatSkill(treeSkill, effectiveLevel)
651           SkillSlotState.fromSkill(combatSkill)
652         case None =>
653           SkillSlotState.fromSkill(CombatSkill.empty)
654     }

662   /** Cost in gold to refund a single skill point */
663   val RefundCostGold: Long = 100

665   /** Allocate a point to a skill */
666   def allocatePoint(state: CombatSkillState, skillId: String): Either[String, CombatSkillState] =
667     if state.availablePoints <= 0 then
668       Left("No skill points available")
669     else
670       SkillTrees.getSkillById(skillId) match
671         case None => Left("Unknown skill")
672         case Some(skill) =>
673           val currentLevel = state.getSkillLevel(skillId)
674           if currentLevel >= skill.maxLevel then
675             Left(s"${skill.name} is already at max level")
676           else
677             // Check unlock requirements
678             skill.unlockRequirement match
679               case Some(req) if state.getSkillLevel(req.skillId) < req.level =>
680                 SkillTrees.getSkillById(req.skillId) match
681                   case Some(reqSkill) =>
682                     Left(s"Requires ${reqSkill.name} level ${req.level}")
683                   case None =>
684                     Left("Invalid requirement")
685               case _ =>
686                 Right(state.copy(
687                   allocatedPoints = state.allocatedPoints.updated(skillId, currentLevel + 1),
688                   availablePoints = state.availablePoints - 1
689                 ))
690 
691   /** Deallocate a point from a skill (refund). Returns new state or error. */
692   def deallocatePoint(state: CombatSkillState, skillId: String): Either[String, CombatSkillState] =
693     SkillTrees.getSkillById(skillId) match
694       case None => Left("Unknown skill")
695       case Some(skill) =>
696         val currentLevel = state.getSkillLevel(skillId)
697         if currentLevel <= 0 then
698           Left(s"${skill.name} has no points to refund")
699         else
700           val newLevel = currentLevel - 1
