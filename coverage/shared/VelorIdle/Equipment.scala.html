84 /** An actual equipment instance with quality, rarity, and affixes */
85 case class EquipmentInstance(
86   defId: String,           // Reference to EquipmentDef.id
87   quality: EquipmentQuality,
88   rarity: EquipmentRarity,
89   affixes: Vector[MagicalAffix] = Vector.empty,
90   instanceId: Long = 0L    // Unique ID for this specific item
91 ) derives ReadWriter:
92 
93   /** Get the definition for this equipment */
94   def definition: Option[EquipmentDef] = EquipmentDefs.byId.get(defId)
95 
96   /** Calculate effective attack damage (base + quality modifier + affixes) */
97   def attackDamage: Int = definition.map { d =>
98     val base = d.baseStats.attackDamage
99     val qualityMod = if quality == EquipmentQuality.Superior then (base * 0.1).toInt else 0
100     val affixMod = affixes.collect { case MagicalAffix.AttackDamageBonus(amt) => amt }.sum
101     base + qualityMod + affixMod
102   }.getOrElse(0)
103 
104   /** Calculate effective defense (base + quality modifier + affixes) */
105   def defense: Int = definition.map { d =>
106     val base = d.baseStats.defense
107     val qualityMod = if quality == EquipmentQuality.Superior then (base * 0.1).toInt else 0
108     val affixMod = affixes.collect { case MagicalAffix.DefenseBonus(amt) => amt }.sum
109     base + qualityMod + affixMod
110   }.getOrElse(0)
111 
112   /** Calculate effective max HP bonus (base + quality modifier + affixes) */
113   def maxHpBonus: Int = definition.map { d =>
114     val base = d.baseStats.maxHpBonus
115     val qualityMod = if quality == EquipmentQuality.Superior then (base * 0.1).toInt else 0
116     val affixMod = affixes.collect { case MagicalAffix.MaxHpBonus(amt) => amt }.sum
117     base + qualityMod + affixMod
118   }.getOrElse(0)
119 
120   /** Calculate max mana bonus from affixes */
121   def maxManaBonus: Int = affixes.collect { case MagicalAffix.MaxManaBonus(amt) => amt }.sum
122 
123   /** Calculate mana regeneration percentage bonus from affixes */
124   def manaRegenBonus: Int = affixes.collect { case MagicalAffix.ManaRegenBonus(pct) => pct }.sum
125 
126   /** Get skill level bonuses from affixes */
127   def skillBonuses: Map[String, Int] =
128     affixes.collect { case MagicalAffix.SkillLevelBonus(treeId, amt) => treeId -> amt }.toMap
129 
130   /** Display name with quality and rarity indicators */
131   def displayName: String = definition.map { d =>
132     val qualityPrefix = EquipmentQuality.icon(quality)
133     val rarityPrefix = EquipmentRarity.icon(rarity)
134     val prefix = (qualityPrefix + rarityPrefix).trim
135     if prefix.isEmpty then d.name else s"$prefix ${d.name}"
136   }.getOrElse("Unknown Equipment")

241 /** Equipment crafting and generation logic */
242 object EquipmentCrafting:
243   // Chances for quality and rarity
244   val SuperiorChance: Double = 0.15  // 15% chance for superior quality
245   val MagicalChance: Double = 0.10   // 10% chance for magical rarity
246 
247   /** Roll for equipment quality (Normal or Superior) */
248   def rollQuality(random: Random): EquipmentQuality =
249     if random.nextDouble() < SuperiorChance then EquipmentQuality.Superior
250     else EquipmentQuality.Normal
251 
252   /** Roll for equipment rarity (Normal or Magical) - independent of quality */
253   def rollRarity(random: Random): EquipmentRarity =
254     if random.nextDouble() < MagicalChance then EquipmentRarity.Magical
255     else EquipmentRarity.Normal
256 
257   /** Generate magical affixes for an equipment piece */
258   def rollAffixes(def_ : EquipmentDef, random: Random): Vector[MagicalAffix] =
259     // Magical items get 1-2 affixes
260     val numAffixes = 1 + random.nextInt(2)
261 
262     val possibleAffixes = def_.slot match
263       case EquipmentSlot.Weapon =>
264         Vector(
265           () => MagicalAffix.AttackDamageBonus(1 + random.nextInt(def_.tier * 2)),
266           () => MagicalAffix.MaxManaBonus(5 + random.nextInt(def_.tier * 5)),
267           () => randomSkillBonus(random)
268         )
269       case EquipmentSlot.Armor =>
270         Vector(
271           () => MagicalAffix.DefenseBonus(1 + random.nextInt(def_.tier * 2)),
272           () => MagicalAffix.MaxHpBonus(5 + random.nextInt(def_.tier * 10)),
273           () => MagicalAffix.MaxManaBonus(5 + random.nextInt(def_.tier * 5)),
274           () => randomSkillBonus(random)
275         )
276 
277     // Pick random unique affixes
278     random.shuffle(possibleAffixes).take(numAffixes).map(_.apply())
279 
280   /** Generate a random skill level bonus affix */
281   private def randomSkillBonus(random: Random): MagicalAffix =
282     val trees = SkillTrees.all
283     val tree = trees(random.nextInt(trees.length))
284     MagicalAffix.SkillLevelBonus(tree.id, 1 + random.nextInt(2))
285 
286   /** Create an equipment instance from a definition */
287   def createEquipment(defId: String, instanceId: Long, random: Random): Option[EquipmentInstance] =
288     EquipmentDefs.byId.get(defId).map { def_ =>
289       val quality = rollQuality(random)
290       val rarity = rollRarity(random)
291       val affixes = if rarity == EquipmentRarity.Magical then rollAffixes(def_, random) else Vector.empty
292       EquipmentInstance(defId, quality, rarity, affixes, instanceId)
293     }
