66 object SkillState:
67   val initial: SkillState = SkillState()
68 
69   /** XP required to go from level N to level N+1.
70     * Formula: 100*level + 10*level² + 0.5*level³ (cubic scaling)
71     * Early levels stay reasonable, but high levels require dramatically more XP.
72     * - Level 1→2: 110 XP
73     * - Level 50→51: 92,500 XP  
74     * - Level 80→81: 326,400 XP
75     * - Level 98→99: 576,626 XP
76     */
77   def xpForLevel(level: Int): Long = 
78     val l = level.toLong
79     (100L * l) + (10L * l * l) + (l * l * l / 2)
80 
81   /** Total XP required to reach a given level from level 1 */
82   def totalXpForLevel(level: Int): Long =
83     if level <= 1 then 0L
84     else (1 until level).map(l => xpForLevel(l)).sum
85 
86   /** Calculate level from total XP */
87   def levelFromXp(totalXp: Long): Int =
88     var level = 1
89     var xpNeeded = 0L
90     while level < 99 && totalXp >= xpNeeded + xpForLevel(level) do
91       xpNeeded += xpForLevel(level)
92       level += 1
93     level
94 
95   /** XP progress within current level (0.0 to 1.0) */
96   def xpProgress(state: SkillState): Double =
97     if state.level >= 99 then 1.0
98     else
99       val currentLevelXp = totalXpForLevel(state.level)
100       val nextLevelXp = totalXpForLevel(state.level + 1)
101       val xpInLevel = state.xp - currentLevelXp
102       val xpNeeded = nextLevelXp - currentLevelXp
103       if xpNeeded <= 0 then 1.0
104       else (xpInLevel.toDouble / xpNeeded).min(1.0).max(0.0)

115 object ActionState:
116   val initial: ActionState = ActionState()
117 
118   /** XP required to go from level N to level N+1, scaled by action tier.
119     * Base formula: 25*level (linear, fast progression)
120     * Tier multiplier: 1 + (actionUnlockLevel / 25)
121     * This keeps action leveling fast but higher tier actions need slightly more.
122     * - Level 1→2: 25 XP (tier 1) to 100 XP (tier 70)
123     * - Level 98→99: 2,450 XP (tier 1) to 9,800 XP (tier 70)
124     */
125   def xpForLevel(level: Int, actionUnlockLevel: Int = 1): Long =
126     val base = 50L * level
127     val tierMultiplier = 1.0 + (actionUnlockLevel / 15.0)
128     (base * tierMultiplier).toLong
129 
130   /** Total XP required to reach a given level from level 1 */
131   def totalXpForLevel(level: Int, actionUnlockLevel: Int = 1): Long =
132     if level <= 1 then 0L
133     else (1 until level).map(l => xpForLevel(l, actionUnlockLevel)).sum
134 
135   /** Calculate level from total XP */
136   def levelFromXp(totalXp: Long, actionUnlockLevel: Int = 1): Int =
137     var level = 1
138     var xpNeeded = 0L
139     while level < 99 && totalXp >= xpNeeded + xpForLevel(level, actionUnlockLevel) do
140       xpNeeded += xpForLevel(level, actionUnlockLevel)
141       level += 1
142     level
143 
144   /** XP progress within current level (0.0 to 1.0) */
145   def xpProgress(state: ActionState, actionUnlockLevel: Int = 1): Double =
146     if state.level >= 99 then 1.0
147     else
148       val currentLevelXp = totalXpForLevel(state.level, actionUnlockLevel)
149       val nextLevelXp = totalXpForLevel(state.level + 1, actionUnlockLevel)
150       val xpInLevel = state.xp - currentLevelXp
151       val xpNeeded = nextLevelXp - currentLevelXp
152       if xpNeeded <= 0 then 1.0
153       else (xpInLevel.toDouble / xpNeeded).min(1.0).max(0.0)
