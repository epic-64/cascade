13   /** Advance game state by elapsed time. Returns updated game and any events that occurred. */
14   def tick(game: VelorIdleGame, currentTime: Long, random: Random = Random): (VelorIdleGame, Vector[GameEvent]) =
15     val elapsed = (currentTime - game.lastTickTime).max(0)
16     val elapsedSeconds = elapsed / 1000.0
17 
18     game.activeAction match
19       case ActiveAction.Idle =>
20         (game.copy(lastTickTime = currentTime), Vector.empty)
21 
22       case ActiveAction.Gathering(skill, action) =>
23         processGatheringTick(game, skill, action, elapsedSeconds, currentTime, random)
24 
25       case ActiveAction.Processing(skill, action) =>
26         processProcessingTick(game, skill, action, elapsedSeconds, currentTime, random)
27 
28       case ActiveAction.EquipmentCrafting(action) =>
29         processEquipmentCraftingTick(game, action, elapsedSeconds, currentTime, random)
30 
31       case ActiveAction.Thieving(action) =>
32         processThievingTick(game, action, elapsedSeconds, currentTime, random)
33 
34       case ActiveAction.Adventure =>
35         // Adventure combat is processed by AdventureCombat.tick
36         AdventureCombat.tick(game, currentTime, random)
37 
38       case ActiveAction.Rest =>
39         // Rest action - regenerate HP and Mana
40         processRestTick(game, elapsedSeconds, currentTime)

42   /** Process rest tick - regenerate HP and Mana for adventure */
43   private def processRestTick(
44     game: VelorIdleGame,
45     elapsedSeconds: Double,
46     currentTime: Long
47   ): (VelorIdleGame, Vector[GameEvent]) =
48     val advState = game.adventureState
49     val maxHp = advState.maxHp
50     val maxMana = advState.maxMana
51 
52     // Use round to avoid losing small increments due to truncation
53     val hpRegen = (AdventureState.HpRegenPerSecond * elapsedSeconds).round.toInt
54     
55     // Mana regen uses accumulator to prevent rounding small values to 0
56     val manaRegenRate = maxMana * AdventureState.RestManaRegenPercent
57     val manaRegenRaw = advState.restManaRegenAccumulator + (manaRegenRate * elapsedSeconds)
58     val manaRegenWhole = manaRegenRaw.toInt
59     val manaRegenFractional = if advState.currentMana + manaRegenWhole >= maxMana then 0.0 else manaRegenRaw - manaRegenWhole
60 
61     val newHp = (advState.currentHp + hpRegen).min(maxHp)
62     val newMana = (advState.currentMana + manaRegenWhole).min(maxMana)
63 
64     val newAdvState = advState.copy(
65       currentHp = newHp, 
66       currentMana = newMana,
67       restManaRegenAccumulator = manaRegenFractional
68     )
69 
70     // Auto-switch to IDLE when fully recovered
71     val isFullyRecovered = newHp >= maxHp && newMana >= maxMana
72     val newActiveAction = if isFullyRecovered then ActiveAction.Idle else game.activeAction
73 
74     val newGame = game.copy(
75       adventureState = newAdvState,
76       activeAction = newActiveAction,
77       lastTickTime = currentTime
78     )
79 
80     (newGame, Vector.empty)
