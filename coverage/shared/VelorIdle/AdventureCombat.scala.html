39   /** Calculate evade chance based on attacker's attack rating vs defender's defense rating.
40     * Formula: evadeChance = defenseRating / (defenseRating + attackRating) * 0.5
41     * This gives a max of 50% evade when defense >> attack, and approaches 0% when attack >> defense.
42     * When attack = defense, evade chance is 25%.
43     */
44   def calculateEvadeChance(attackRating: Int, defenseRating: Int): Double =
45     if attackRating + defenseRating <= 0 then 0.0
46     else (defenseRating.toDouble / (defenseRating + attackRating)) * 0.5
47 
48   /** Check if an attack is evaded */
49   def rollEvade(attackRating: Int, defenseRating: Int, random: Random): Boolean =
50     random.nextDouble() < calculateEvadeChance(attackRating, defenseRating)

118   /** Process one tick of combat. Returns updated game and events.
119     * 
120     * State machine transitions:
121     * - Fighting: Process combat (attacks, skills, damage). If killing blow dealt -> WaitingForProjectiles
122     * - WaitingForProjectiles: Only process pending damage. When all projectiles land -> Victory/Defeat
123     * - Victory: Wait for timer, then spawn next enemy -> Fighting
124     * - Defeat: Combat ends, no more processing
125     */
126   def tick(
127     game: VelorIdleGame,
128     currentTime: Long,
129     random: Random = Random
130   ): (VelorIdleGame, Vector[GameEvent]) =
131     val (updatedGame, events) = game.adventureState.combatState match
132       case None =>
133         (game, Vector.empty)
134         
135       case Some(combat) =>
136         combat.phase match
137           case CombatPhase.Fighting =>
138             processFightingPhase(game, combat, currentTime, random)
139             
140           case CombatPhase.WaitingForProjectiles(victorIsPlayer) =>
141             processWaitingPhase(game, combat, currentTime, victorIsPlayer, random)
142             
143           case CombatPhase.Victory(spawnNextEnemyAt) =>
144             processVictoryPhase(game, combat, currentTime, spawnNextEnemyAt)
145             
146           case CombatPhase.Defeat =>
147             // Combat ended - no more processing
148             (game, Vector.empty)
149     
150     // Always update lastTickTime
151     (updatedGame.copy(lastTickTime = currentTime), events)

153   /** Fighting phase: full combat processing */
154   private def processFightingPhase(
155     game: VelorIdleGame,
156     combat: CombatState,
157     currentTime: Long,
158     random: Random
159   ): (VelorIdleGame, Vector[GameEvent]) =
160     val elapsedSeconds = (currentTime - game.lastTickTime).max(0) / 1000.0
161 
162     // Run all combat systems
163     val (processedCombat, combatEvents) = runCombatSystems(combat, currentTime, elapsedSeconds, game.adventureState, random)
164 
165     // Check if a killing blow was just dealt (but projectiles may still be in flight)
166     val killingBlowToEnemy = processedCombat.isEnemyDead
167     val killingBlowToPlayer = processedCombat.isPlayerDead
168     
169     if killingBlowToEnemy || killingBlowToPlayer then
170       // Transition to WaitingForProjectiles if there are pending projectiles
171       if processedCombat.hasPendingDamage then
172         val newPhase = CombatPhase.WaitingForProjectiles(victorIsPlayer = killingBlowToEnemy)
173         val updatedCombat = processedCombat.copy(phase = newPhase)
174         finalizeCombatState(updatedCombat, game, combatEvents)
175       else
176         // No pending projectiles - go directly to Victory/Defeat
177         if killingBlowToEnemy then
178           handleVictory(processedCombat, game, currentTime, random)
179         else
180           handleDefeat(processedCombat, game, combatEvents)
181     else
182       finalizeCombatState(processedCombat, game, combatEvents)

184   /** WaitingForProjectiles phase: only process pending damage, no new actions */
185   private def processWaitingPhase(
186     game: VelorIdleGame,
187     combat: CombatState,
188     currentTime: Long,
189     victorIsPlayer: Boolean,
190     random: Random
191   ): (VelorIdleGame, Vector[GameEvent]) =
192     // Only process pending damage - no new attacks or skills
193     val (afterPending, pendingDetails) = processPendingDamage(combat, currentTime)
194     val (withEvents, combatEvents) = finalizeEvents(afterPending, pendingDetails)
195     
196     // Check if all projectiles have landed
197     if !withEvents.hasPendingDamage then
198       // All projectiles landed - transition to Victory or Defeat
199       if victorIsPlayer then
200         handleVictory(withEvents, game, currentTime, random)
201       else
202         handleDefeat(withEvents, game, combatEvents)
203     else
204       // Still waiting for projectiles
205       finalizeCombatState(withEvents, game, combatEvents)

207   /** Victory phase: wait for timer, then spawn next enemy */
208   private def processVictoryPhase(
209     game: VelorIdleGame,
210     combat: CombatState,
211     currentTime: Long,
212     spawnNextEnemyAt: Long
213   ): (VelorIdleGame, Vector[GameEvent]) =
214     if currentTime >= spawnNextEnemyAt then
215       // Timer expired - spawn next enemy
216       restartCombat(game, currentTime) match
217         case Right(restarted) =>

219         case Left(_) =>

221           val clearedState = game.adventureState.copy(inCombat = false, combatState = None)

227   /** Run all combat systems: casting, DoTs, auto-attacks, regen, etc. 
228     * Only called during Fighting phase.
229     */
230   private def runCombatSystems(
231     combat: CombatState,
232     currentTime: Long,
233     elapsedSeconds: Double,
234     advState: AdventureState,
235     random: Random
236   ): (CombatState, Vector[CombatEvent]) =
237     var c = combat
238     var details = Vector.empty[CombatEventDetail]
239 
240     // 1. Process casting
241     c.castingSkill match
242       case Some(casting) if casting.isComplete(currentTime) =>
243         val (newCombat, castDetails) = executeCastedSkill(c, casting.slotIndex, casting.skill, currentTime)
244         c = newCombat.copy(castingSkill = None)
245         details ++= castDetails
246       case _ => ()
247 
248     // 2. Process pending damage (projectiles landing)
249     val (afterPending, pendingDetails) = processPendingDamage(c, currentTime)
250     c = afterPending
251     details ++= pendingDetails
252 
253     // 3. Process DoTs (only if both combatants still alive)
254     if !c.isEnemyDead && !c.isPlayerDead then
255       val (afterDoTs, dotDetails) = processDoTs(c, currentTime)
256       c = afterDoTs
257       details ++= dotDetails
258 
259     // 4. Process auto-attacks (only if both combatants still alive)
260     if !c.isEnemyDead && !c.isPlayerDead then
261       val (afterAutos, autoDetails) = processAutoAttacks(c, currentTime, advState, random)
262       c = afterAutos
263       details ++= autoDetails
264 
265     // 5. Mana regen (use floating point accumulator for precision)
266     val manaRegen = advState.manaRegenPerSecond * elapsedSeconds
267     val newMana = (c.playerMana.toDouble + c.manaRegenAccumulator + manaRegen)
268     val wholeMana = newMana.toInt.min(c.playerMaxMana)
269     val fractionalMana = if wholeMana >= c.playerMaxMana then 0.0 else newMana - wholeMana
270     c = c.copy(playerMana = wholeMana, manaRegenAccumulator = fractionalMana)
271 
272     // 6. Process chain windows
273     c = processChainWindows(c, currentTime)
274 
275     // 7. Expire timed effects
276     c = expireTimedEffects(c, currentTime)
277 
278     finalizeEvents(c, details)

283   /** Handle victory: grant rewards and transition to Victory phase */
284   private def handleVictory(
285     combat: CombatState,
286     game: VelorIdleGame,
287     currentTime: Long,
288     random: Random
289   ): (VelorIdleGame, Vector[GameEvent]) =
290     val enemy = combat.enemy
291     var events = Vector.empty[GameEvent]
292     var g = game
293 
294     // Grant XP
295     val skillState = g.skills.getOrElse(Skill.Adventure, SkillState.initial)
296     val oldLevel = skillState.level
297     val newXp = skillState.xp + enemy.xpReward
298     val newLevel = SkillState.levelFromXp(newXp)
299     g = g.copy(skills = g.skills.updated(Skill.Adventure, skillState.copy(xp = newXp, level = newLevel)))
300     events = events :+ GameEvent.XpGained(Skill.Adventure, enemy.xpReward)
301     
302     // Check for level up and award skill points
303     if newLevel > oldLevel then
304       events = events :+ GameEvent.LevelUp(Skill.Adventure, newLevel)
305       val (updatedGame, skillPointsAwarded) = VelorIdleLogic.checkAndAwardAdventureSkillPoints(g, oldLevel)
306       g = updatedGame
307       if skillPointsAwarded > 0 then
308         events = events :+ GameEvent.SkillPointsGained(skillPointsAwarded)
309 
310     // Grant gold
311     val goldAmount = enemy.goldReward._1 + random.nextInt(enemy.goldReward._2 - enemy.goldReward._1 + 1)
312     g = g.copy(gold = g.gold + goldAmount)
313     events :+= GameEvent.GoldGained(goldAmount)
314 
315     // Grant loot
316     enemy.lootTable.foreach { case (item, chance) =>
317       if random.nextDouble() < chance then
318         val (newInv, _) = g.inventory.addItem(item, 1)
319         g = g.copy(inventory = newInv)
320         events :+= GameEvent.ItemGained(item, 1)
321     }
322 
323     events :+= GameEvent.AdventureEnemyDefeated(enemy.id)
324 
325     // Transition to Victory phase - next enemy spawns after delay
326     val victoryCombat = combat.copy(
327       phase = CombatPhase.Victory(spawnNextEnemyAt = currentTime + VictoryDisplayMs)
328     )
329     val newAdvState = g.adventureState.copy(
330       combatState = Some(victoryCombat),
331       currentHp = combat.playerCurrentHp,
332       currentMana = combat.playerMana
333     )
334     (g.copy(adventureState = newAdvState), events)

369   /** Process a single DoT list, returning updated HP, remaining DoTs, and events */
370   private def tickDoTs(
371     dots: Vector[ActiveDoT],
372     currentHp: Int,
373     currentTime: Long,
374     makeEvent: (Int, String) => CombatEventDetail
375   ): (Int, Vector[ActiveDoT], Vector[CombatEventDetail]) =
376     var hp = currentHp
377     var events = Vector.empty[CombatEventDetail]
378     val remaining = dots.flatMap { dot =>
379       if currentTime >= dot.lastTickTime + dot.tickIntervalMs then
380         hp -= dot.damagePerTick
381         events :+= makeEvent(dot.damagePerTick, dot.name)
382         val ticksLeft = dot.ticksRemaining - 1
383         if ticksLeft > 0 then Some(dot.copy(ticksRemaining = ticksLeft, lastTickTime = currentTime))
384         else None
385       else Some(dot)
386     }
387     (hp.max(0), remaining, events)

417   /** Process pending damage - projectiles that have landed */
418   private def processPendingDamage(
419     combat: CombatState,
420     currentTime: Long
421   ): (CombatState, Vector[CombatEventDetail]) =
422     var c = combat
423     var events = Vector.empty[CombatEventDetail]
424     
425     // Process all pending skills (includes auto-attacks and player skills)
426     val (landedSkills, stillPendingSkills) = c.pendingSkills.partition(_.landsAt <= currentTime)
427     c = c.copy(pendingSkills = stillPendingSkills)
428     
429     landedSkills.foreach { ps =>
430       ps.pendingType match
431         case PendingType.PlayerAutoAttack =>
432           if ps.evaded then
433             events :+= CombatEventDetail.EnemyEvaded
434           else
435             c = c.copy(enemyCurrentHp = (c.enemyCurrentHp - ps.skill.damage).max(0))
436             // Note: PlayerAutoAttack event was already emitted when attack fired
437             
438         case PendingType.EnemyAutoAttack =>
439           if ps.evaded then
440             events :+= CombatEventDetail.PlayerEvaded
441           else
442             val (newHp, newShield, shieldBroken) = applyDamageWithShield(c.playerCurrentHp, c.playerShield, ps.skill.damage, currentTime)
443             c = c.copy(playerCurrentHp = newHp.max(0), playerShield = newShield)
444             // Note: EnemyAutoAttack event was already emitted when attack fired
445             if shieldBroken then events :+= CombatEventDetail.ShieldBroken
446             
447         case PendingType.PlayerSkill(slotIndex) =>
448           val (newCombat, skillEvents) = applySkillEffectsOnLanding(c, slotIndex, ps.skill, currentTime, ps.damageBuffPercent)
449           c = newCombat
450           events ++= skillEvents
451     }
452     
453     (c, events)

455   private def processAutoAttacks(
456     combat: CombatState,
457     currentTime: Long,
458     advState: AdventureState,
459     random: Random
460   ): (CombatState, Vector[CombatEventDetail]) =
461     var c = combat
462     var events = Vector.empty[CombatEventDetail]
463 
464     // Player auto-attack (disabled while casting)
465     val canPlayerAutoAttack = !c.isCasting && 
466                               currentTime >= c.lastPlayerAutoAttack + BaseAutoAttackSpeedMs
467     
468     if canPlayerAutoAttack then
469       val evaded = rollEvade(advState.attackRating, c.enemy.defenseRating, random)
470       c = c.copy(lastPlayerAutoAttack = currentTime, playerDamageBuff = None)
471       
472       val damage = if evaded then 0 else
473         val baseDamage = advState.attackDamageMin + random.nextInt((advState.attackDamageMax - advState.attackDamageMin + 1).max(1))
474         applyDamageBuff(baseDamage, combat.playerDamageBuff)
475       
476       // Create auto-attack skill with calculated damage
477       val autoAttackSkill = CombatSkill(
478         id = "player_auto_attack",
479         name = "Auto Attack",
480         icon = "⚔️",
481         description = "Basic attack",
482         manaCost = 0,
483         cooldownMs = 0,
484         damage = damage
485       )
486       
487       // Emit event immediately (like regular skills)
488       events :+= CombatEventDetail.PlayerAutoAttack(damage)
489       
490       // Queue pending skill to land after flight time
491       val pending = PendingSkill(
492         pendingType = PendingType.PlayerAutoAttack,
493         skill = autoAttackSkill,
494         landsAt = currentTime + ProjectileFlightTimeMs,
495         evaded = evaded
496       )
497       c = c.copy(pendingSkills = c.pendingSkills :+ pending)
498 
499     // Enemy auto-attack (only if not stunned/frozen and player not dead)
500     val canEnemyAttack = !c.enemyStun.exists(_.endsAt > currentTime) &&
501                          !c.enemyFreeze.exists(_.endsAt > currentTime) &&
502                          !c.isPlayerDead &&
503                          currentTime >= c.lastEnemyAutoAttack + c.enemy.attackSpeedMs
504     
505     if canEnemyAttack then
506       val evaded = rollEvade(c.enemy.attackRating, advState.defenseRating, random)
507       c = c.copy(lastEnemyAutoAttack = currentTime)
508       
509       val damage = if evaded then 0 else c.enemy.attackDamage
510       
511       // Create enemy auto-attack skill
512       val enemyAutoAttackSkill = CombatSkill(
513         id = "enemy_auto_attack",
514         name = "Enemy Attack",
515         icon = "💥",
516         description = "Enemy basic attack",
517         manaCost = 0,
518         cooldownMs = 0,
519         damage = damage
520       )
521       
522       // Emit event immediately
523       events :+= CombatEventDetail.EnemyAutoAttack(damage)
524       
525       // Queue pending skill
526       val pending = PendingSkill(
527         pendingType = PendingType.EnemyAutoAttack,
528         skill = enemyAutoAttackSkill,
529         landsAt = currentTime + ProjectileFlightTimeMs,
530         evaded = evaded
531       )
532       c = c.copy(pendingSkills = c.pendingSkills :+ pending)
533 
534     (c, events)

580   private def validateAndExecuteSkill(
581     game: VelorIdleGame,
582     combat: CombatState,
583     slotIndex: Int,
584     currentTime: Long
585   ): Either[String, VelorIdleGame] =
586     if slotIndex < 0 || slotIndex >= combat.skillSlots.length then
587       return Left(s"Invalid skill slot: $slotIndex")
588 
589     val slot = combat.skillSlots(slotIndex)
590     val skill = slot.currentSkill
591     val isChainSkill = slot.isInChainWindow(currentTime) && skill.id != slot.baseSkill.id
592 
593     // Validation
594     if skill.id == "empty" then Left("No skill in this slot")
595     else if combat.isCasting then Left("Already casting a skill")
596     else if combat.isOnGlobalCooldown(currentTime) then
597       Left(f"Global cooldown (${combat.globalCooldownRemainingMs(currentTime) / 1000.0}%.1fs)")
