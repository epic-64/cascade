126   private def handleClientMessage(channel: cask.WsChannelActor, lobbyId: String, msg: ClientMessage): Unit =
127     val actualLobbyId = Option(channelToLobby.get(channel)).getOrElse(lobbyId)
128 
129     msg match
130       case ClientMessage.CreateLobby(playerName, apiKey, gameMode) =>
131         handleCreateLobby(channel, playerName, apiKey, gameMode)
132 
133       case ClientMessage.JoinLobby(joinLobbyId, playerName) =>
134         handleJoinLobby(channel, joinLobbyId, playerName)
135 
136       case ClientMessage.RejoinLobby(rejoinLobbyId, playerId) =>
137         handleRejoinLobby(channel, rejoinLobbyId, playerId)
138 
139       case ClientMessage.StartGame() =>
140         handleStartGame(actualLobbyId)
141 
142       case ClientMessage.SubmitDrawing(imageData) =>
143         handleSubmitDrawing(channel, actualLobbyId, imageData)
144 
145       case ClientMessage.SubmitVote(playerNameVotedFor) =>
146         handleSubmitVote(channel, actualLobbyId, playerNameVotedFor)
147 
148       case ClientMessage.NextRound() =>
149         handleNextRound(actualLobbyId)
150 
151       case ClientMessage.Ping() =>
152         // Keepalive ping - no action needed, just keeps the connection alive
153         logger.debug(s"Received keepalive ping for lobby $actualLobbyId")

203   private def handleRejoinLobby(channel: cask.WsChannelActor, lobbyId: String, playerId: String): Unit =
204     // Use trait's handleRejoinRequest for the core rejoin logic
205     handleRejoinRequest(channel, lobbyId, playerId)
206 
207     // If rejoin was successful, send additional game-specific state
208     // Check if channel is now registered to this lobby
209     if Option(channelToLobby.get(channel)).contains(lobbyId) then
210       getGame(lobbyId).foreach: lobby =>
211         // Send current lobby state
212         sendToClient(channel, ServerMessage.LobbyUpdate(lobby))
213 
214         lobby.status match
215           case LobbyStatus.Drawing =>
216             // Send current prompt if in drawing phase
217             lobby.currentPrompt.foreach: prompt =>
218               sendToClient(channel, ServerMessage.PromptAnnounced(prompt))
219 
220           case LobbyStatus.RevealingDrawings | LobbyStatus.RevealingCaptions |
221               LobbyStatus.RevealingAIWinner | LobbyStatus.Voting | LobbyStatus.Results =>
222             // Send drawings data so gallery can be populated
223             lobby.currentPrompt.foreach: prompt =>
224               val drawings = lobby.drawings.values.toSeq
225               sendToClient(channel, ServerMessage.DrawingsRevealed(drawings, prompt))
226 
227               // Also send any captions that have been revealed
228               drawings.foreach: drawing =>
229                 drawing.caption.foreach: caption =>
230                   sendToClient(channel, ServerMessage.CaptionRevealed(drawing.playerName, caption))
231 
232               // Send AI winner if already revealed
233               Option(aiWinners.get(lobbyId)).foreach: winnerName =>
234                 sendToClient(channel, ServerMessage.AIVoteRevealed(winnerName))
235 
236               // If in voting phase, notify client
237               if lobby.status == LobbyStatus.Voting then
238                 // Calculate remaining time approximately
239                 val elapsed = lobby.timerStartTime.map(t => (System.currentTimeMillis() - t) / 1000).getOrElse(0L)
240                 val remaining = Math.max(0, DrawingGame.votingTimeSeconds - elapsed.toInt)
241                 sendToClient(channel, ServerMessage.VotingStarted(remaining))
242 
243                 // Send current vote counts
244                 val voteCounts = DrawingGame.tallyVotes(lobby)
245                 sendToClient(channel, ServerMessage.VoteUpdate(voteCounts))
246 
247               // If in results phase, send round complete
248               if lobby.status == LobbyStatus.Results then
249                 val voteCounts = DrawingGame.tallyVotes(lobby)
250                 val playerWinner = voteCounts.maxByOption(_._2).map(_._1)
251                 val aiWinnerName = Option(aiWinners.get(lobbyId))
252                 val result = RoundResult(aiWinnerName, playerWinner, voteCounts)
253                 sendToClient(channel, ServerMessage.RoundComplete(result))
254 
255           case _ => () // Waiting, CollectingDrawings - no extra state needed

338   private def transitionTo(lobbyId: String, newStatus: LobbyStatus): Unit =
339     lobbies.get(lobbyId) match
340       case null =>
341         logger.warn(s"Cannot transition non-existent lobby $lobbyId to $newStatus")
342       case (lobby, apiKey) =>
343         logger.info(s"Lobby $lobbyId: ${lobby.status} -> $newStatus")
344 
345         newStatus match
346           case LobbyStatus.Waiting =>
347             val updated = DrawingGame.resetForNextRound(lobby)
348             lobbies.put(lobbyId, (updated, apiKey))
349             broadcastLobbyUpdate(lobbyId)
350 
351           case LobbyStatus.GeneratingPrompt =>
352             // Update status to show loading state
353             val updated = lobby.copy(status = LobbyStatus.GeneratingPrompt)
354             lobbies.put(lobbyId, (updated, apiKey))
355             broadcastLobbyUpdate(lobbyId)
356             broadcast(lobbyId, ServerMessage.GeneratingPrompt())
357             // Start the async prompt generation
358             startAdvancedDrawingPhase(lobbyId, updated, apiKey)
359 
360           case LobbyStatus.Drawing =>
361             val needsGeneration = lobby.gameMode != GameMode.SingleWord && lobby.status != LobbyStatus.GeneratingPrompt
362             if needsGeneration then
363               // Advanced modes: go through GeneratingPrompt first
364               transitionTo(lobbyId, LobbyStatus.GeneratingPrompt)
365             else
366               // SingleWord mode OR coming from GeneratingPrompt: start drawing
367               val updated = if lobby.status == LobbyStatus.GeneratingPrompt then
368                 // Already have prompt set by startAdvancedDrawingPhase
369                 lobby.copy(
370                   status = LobbyStatus.Drawing,
371                   currentRound = lobby.currentRound + 1,
372                   drawings = Map.empty,
373                   votes = Map.empty,
374                   timerStartTime = Some(System.currentTimeMillis())
375                 )
376               else
377                 DrawingGame.startDrawingPhase(lobby)
378               lobbies.put(lobbyId, (updated, apiKey))
379               updated.currentPrompt.foreach: prompt =>
380                 broadcast(lobbyId, ServerMessage.PromptAnnounced(prompt))
381               broadcastLobbyUpdate(lobbyId)
382               startTimer(
383                 lobbyId,
384                 DrawingGame.drawingTimeSeconds,
385                 LobbyStatus.CollectingDrawings,
386                 seconds => Some(ServerMessage.DrawingTimerUpdate(seconds))
387               )
388 
389           case LobbyStatus.CollectingDrawings =>
390             val updated = lobby.copy(status = LobbyStatus.CollectingDrawings)
391             lobbies.put(lobbyId, (updated, apiKey))
392             broadcastLobbyUpdate(lobbyId)
393             // Brief pause (1 second) to collect any last-moment submissions
394             startTimer(lobbyId, 1, LobbyStatus.RevealingDrawings)
395 
396           case LobbyStatus.RevealingDrawings =>
397             val updated = lobby.copy(status = LobbyStatus.RevealingDrawings)
398             lobbies.put(lobbyId, (updated, apiKey))
399             broadcastLobbyUpdate(lobbyId)
400             // Send drawings (without captions)
401             val drawingsWithoutCaptions = updated.drawings.values.map(_.copy(caption = None)).toSeq
402             broadcast(
403               lobbyId,
404               ServerMessage.DrawingsRevealed(drawingsWithoutCaptions, lobby.currentPrompt.getOrElse(""))
405             )
406             // Start AI captioning in background, then transition
407             startCaptioning(lobbyId)
408 
409           case LobbyStatus.RevealingCaptions =>
410             val updated = lobby.copy(status = LobbyStatus.RevealingCaptions)
411             lobbies.put(lobbyId, (updated, apiKey))
412             broadcastLobbyUpdate(lobbyId)
413             // Reveal next caption or move to AI winner
414             revealNextCaption(lobbyId)
415 
416           case LobbyStatus.RevealingAIWinner =>
417             val updated = lobby.copy(status = LobbyStatus.RevealingAIWinner)
418             lobbies.put(lobbyId, (updated, apiKey))
419             broadcastLobbyUpdate(lobbyId)
420             // Get AI winner and announce
421             startAIWinnerSelection(lobbyId)
422 
423           case LobbyStatus.Voting =>
424             val updated = DrawingGame.startVoting(lobby)
425             lobbies.put(lobbyId, (updated, apiKey))
426             broadcastLobbyUpdate(lobbyId)
427             broadcast(lobbyId, ServerMessage.VotingStarted(DrawingGame.votingTimeSeconds))
428             startTimer(
429               lobbyId,
430               DrawingGame.votingTimeSeconds,
431               LobbyStatus.Results,
432               seconds => Some(ServerMessage.VotingTimerUpdate(seconds))
433             )
434 
435           case LobbyStatus.Results =>
436             completeRound(lobbyId)

446   private def startTimer(
447       lobbyId: String,
448       durationSeconds: Int,
449       nextStatus: LobbyStatus,
450       timerMessage: Int => Option[ServerMessage] = _ => None
451   ): Unit =
452     cancelTimer(lobbyId)
453 
454     val futures = new java.util.ArrayList[ScheduledFuture[?]]()
455 
456     // Schedule countdown updates (only if timerMessage is provided)
457     (1 to durationSeconds).foreach: i =>
458       val secondsRemaining = durationSeconds - i
459       timerMessage(secondsRemaining).foreach: msg =>
460         val future = timerScheduler.schedule(
461           (() => broadcast(lobbyId, msg)): Runnable,
462           i.toLong,
463           TimeUnit.SECONDS
464         )
465         futures.add(future)
466 
467     // Schedule the transition to next state
468     val transitionTask: Runnable = () => transitionTo(lobbyId, nextStatus)
469     val transitionFuture = timerScheduler.schedule(transitionTask, durationSeconds.toLong, TimeUnit.SECONDS)
470     futures.add(transitionFuture)
471 
472     activeTimers.put(lobbyId, futures)

506   private def startCaptioning(lobbyId: String): Unit =
507     lobbies.get(lobbyId) match
508       case null => ()
509       case (lobby, apiKey) =>
510         if lobby.drawings.isEmpty then
511           // No drawings to caption, skip to voting
512           transitionTo(lobbyId, LobbyStatus.Voting)
513         else
514           // Caption all drawings in parallel
515           val captioningFutures = lobby.drawings.map: (playerId, drawing) =>
516             OpenAIClient.captionImage(apiKey, drawing.imageData).map: caption =>
517               (playerId, drawing.playerName, caption)
518 
519           Future.sequence(captioningFutures).map: captionResults =>
520             // Store captions in lobby
521             var currentLobby = lobby
522             captionResults.foreach: (playerId, _, caption) =>
523               currentLobby = DrawingGame.addCaption(currentLobby, playerId, caption)
524             lobbies.put(lobbyId, (currentLobby, apiKey))
525 
526             // Store pending captions for sequential reveal
527             pendingCaptions.put(lobbyId, captionResults.toList)
528 
529             // Transition to caption reveal phase
530             transitionTo(lobbyId, LobbyStatus.RevealingCaptions)
531           .recover:
532             case ex =>
533               logger.error(s"Error captioning images: ${ex.getMessage}", ex)
534               broadcast(lobbyId, ServerMessage.ErrorMessage("AI captioning failed"))
535               // Skip to voting anyway
536               transitionTo(lobbyId, LobbyStatus.Voting)

550       case _ =>
551         // All captions revealed, move to AI winner

563         val captions = lobby.drawings.values.map(d => d.playerName -> d.caption.getOrElse("")).toMap

578             .recover:
579               case ex =>
