218   private def startCountdown(gameId: String, timeLimitSeconds: Int): Unit =
219     // Cancel any existing timer
220     stopRoundTimer(gameId)
221 
222     var countdownRemaining = shared.TugOfWar.TugOfWar.CountdownSeconds

224     // Send initial countdown value
225     broadcastCountdownUpdate(gameId, countdownRemaining)

228       def run(): Unit =
