52   // Internal helper for making OpenAI requests (used by RealOpenAIClient)
53   private[server] def makeOpenAIRequest(
54       url: String,
55       apiKey: String,
56       requestBody: ujson.Obj
57   )(using ec: ExecutionContext): Future[ujson.Value] =
58     Future:
59       val connection = java.net.URI.create(url).toURL.openConnection().asInstanceOf[java.net.HttpURLConnection]
60 
61       Try:
62         connection.setRequestMethod("POST")
63         connection.setRequestProperty("Content-Type", "application/json")
64         connection.setRequestProperty("Authorization", s"Bearer $apiKey")
65         connection.setDoOutput(true)
66         connection.setConnectTimeout(30000) // 30 seconds
67         connection.setReadTimeout(30000)
68 
69         // Write request body
70         val outputStream = connection.getOutputStream
71         outputStream.write(requestBody.toString.getBytes("UTF-8"))
72         outputStream.flush()
73         outputStream.close()
74 
75         // Read response
76         val responseCode = connection.getResponseCode
77         val inputStream = if responseCode >= 400 then connection.getErrorStream else connection.getInputStream
78         val response = scala.io.Source.fromInputStream(inputStream, "UTF-8").mkString
79         inputStream.close()
80 
81         if responseCode != 200 then
82           logger.error(s"OpenAI API error (status $responseCode): $response")
83           throw new RuntimeException(s"OpenAI API request failed with status $responseCode")
84 
85         ujson.read(response)
86       .recover:
87         case ex: Exception =>
88           logger.error(s"OpenAI API request failed: ${ex.getMessage}", ex)
89           throw ex
90       .get
