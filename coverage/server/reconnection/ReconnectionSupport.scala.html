6 /** Mixin trait providing standardized reconnection handling for game servers.
7   *
8   * Games implement the abstract methods to provide game-specific behavior, and get the reconnection logic for free.
9   *
10   * @tparam Player
11   *   The player type (must extend PlayerConnection)
12   * @tparam Game
13   *   The game/lobby state type
14   */
15 trait ReconnectionSupport[Player <: PlayerConnection, Game]:

18   protected def logger: Logger

30   protected def disconnectPlayer(player: Player): Player

53   /** Handle a rejoin request from a client.
54     *
55     * This is the main entry point that games call when receiving a rejoin message.
56     *
57     * @param channel
58     *   The WebSocket channel
59     * @param gameId
60     *   The game to rejoin
61     * @param playerId
62     *   The player attempting to rejoin
63     */
64   final def handleRejoinRequest(
65       channel: cask.WsChannelActor,
66       gameId: String,
67       playerId: String
68   ): Unit =
69     getGame(gameId) match
70       case Some(game) =>
71         getPlayers(game).get(playerId) match
72           case Some(player) if PlayerConnectionOps.canRejoin(player, gracePeriodMs) =>
73             performRejoin(channel, gameId, playerId, player, game)
74           case Some(_) =>
75             logger.info(s"Player $playerId cannot rejoin game $gameId - grace period expired")
76             sendRejoinFailure(channel, "Session expired")
77           case None =>
78             logger.info(s"Player $playerId not found in game $gameId")
79             sendRejoinFailure(channel, "Player not found")
80       case None =>
81         logger.info(s"Game $gameId not found for rejoin request from $playerId")
82         sendRejoinFailure(channel, "Game not found")
83 
84   /** Perform the actual rejoin operation.
85     */
86   private def performRejoin(
87       channel: cask.WsChannelActor,
88       gameId: String,
89       playerId: String,
90       player: Player,
91       game: Game
92   ): Unit =
93     val reconnectedPlayer = reconnectPlayer(player)
94     val updatedGame = withUpdatedPlayer(game, playerId, reconnectedPlayer)
95 
96     registerPlayerChannel(channel, gameId, playerId)
97     saveGame(gameId, updatedGame)
98 
99     logger.info(s"Player $playerId rejoined game $gameId")
100 
101     sendRejoinSuccess(channel, playerId, gameId)
102     broadcastGameState(gameId)
103 
104   /** Handle a player disconnection.
105     *
106     * Marks the player as disconnected rather than removing them, allowing them to rejoin within the grace period.
107     *
108     * @param gameId
109     *   The game the player disconnected from
110     * @param playerId
111     *   The player who disconnected
112     */
113   final def handleDisconnection(gameId: String, playerId: String): Unit =
114     getGame(gameId).foreach: game =>
115       getPlayers(game).get(playerId).foreach: player =>
116         val disconnectedPlayer = disconnectPlayer(player)
117         val updatedGame = withUpdatedPlayer(game, playerId, disconnectedPlayer)
118         saveGame(gameId, updatedGame)
119         logger.info(s"Player $playerId disconnected from game $gameId (can rejoin within ${gracePeriodMs}ms)")
120         broadcastGameState(gameId)
121 
122   /** Clean up players who have exceeded the grace period.
123     *
124     * @param game
125     *   The game to clean up
126     * @param removePlayer
127     *   Function to remove a player from the game
128     * @return
129     *   Updated game with expired players removed
130     */
131   final def cleanupExpiredPlayers(
132       game: Game,
133       removePlayer: (Game, String) => Game
134   ): Game =
135     val expiredPlayerIds = getPlayers(game).collect:
136       case (playerId, player) if PlayerConnectionOps.isGracePeriodExpired(player, gracePeriodMs) =>
137         playerId
138 
139     expiredPlayerIds.foldLeft(game): (g, playerId) =>
140       logger.info(s"Removing expired player $playerId")
141       removePlayer(g, playerId)
