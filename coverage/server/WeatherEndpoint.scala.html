9   // WMO Weather interpretation codes (WW)
10   // https://open-meteo.com/en/docs
11   private def weatherCodeToDescription(code: Int): String = code match
12     case 0            => "Clear sky"
13     case 1 | 2 | 3    => "Partly cloudy"
14     case 45 | 48      => "Foggy"
15     case 51 | 53 | 55 => "Drizzle"
16     case 61 | 63 | 65 => "Rain"
17     case 71 | 73 | 75 => "Snow"
18     case 77           => "Snow grains"
19     case 80 | 81 | 82 => "Rain showers"
20     case 85 | 86      => "Snow showers"
21     case 95           => "Thunderstorm"
22     case 96 | 99      => "Thunderstorm with hail"
23     case _            => "Unknown"
24 
25   def getWeather(city: String)(using client: WeatherClient): ujson.Value =
26     client.getWeather(city) match
27       case Success(data) =>
28         // Extract relevant info from the Open-Meteo API response
29         val current = data("current")
30         val weatherCode = current("weather_code").num.toInt
31         val tempC = current("temperature_2m").num
32         val tempF = (tempC * 9 / 5) + 32
33 
34         ujson.Obj(
35           "city" -> city,
36           "temperature_c" -> tempC.toString,
37           "temperature_f" -> f"$tempF%.1f",
38           "condition" -> weatherCodeToDescription(weatherCode),
39           "humidity" -> current("relative_humidity_2m").num.toString,
40           "feelsLike_c" -> current("apparent_temperature").num.toString
41         )
42 
43       case Failure(ex) =>
44         logger.error(s"Failed to fetch weather for $city: ${ex.getMessage}", ex)
45         ujson.Obj(
46           "error" -> "Failed to fetch weather data",
47           "message" -> ex.getMessage
48         )
