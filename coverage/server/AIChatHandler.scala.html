37   private def handleMessage(channel: cask.WsChannelActor, data: String): Unit =
38     // First try to parse as a special GenerateWithContext message
39     Try(ujson.read(data)) match
40       case Success(json) if json.obj.get("$type").exists(_.str == "GenerateWithContext") =>
41         handleGenerateWithContext(channel, json)
42       case _ =>
43         // Otherwise parse as regular ClientMessage
44         Try(read[ClientMessage](data)) match
45           case Success(msg) => processMessage(channel, msg)
46           case Failure(ex) =>
47             logger.error(s"[AIChat] Failed to parse message: ${ex.getMessage}")
48             sendMessage(channel, ServerMessage.ErrorMessage(s"Invalid message format: ${ex.getMessage}"))

50   private def handleGenerateWithContext(channel: cask.WsChannelActor, json: ujson.Value): Unit =
51     json.obj.get("apiKey").map(_.str).filter(_.nonEmpty) match
52       case Some(apiKey) =>
53         Try:
54           val messages = read[Seq[ChatMessage]](json("messages"))
55           val model = json.obj.get("model").map(_.str).getOrElse(AIChat.defaultModel)
56           // Client already added user message locally, just generate response
57           generateResponse(channel, apiKey, messages, model)
58         .recover:
59           case ex: Exception =>
60             logger.error(s"[AIChat] Failed to parse context: ${ex.getMessage}")
61             sendMessage(channel, ServerMessage.ErrorMessage(s"Failed to parse context: ${ex.getMessage}"))
62       case None =>
63         sendMessage(channel, ServerMessage.ErrorMessage("Please include your API key"))

65   private def processMessage(channel: cask.WsChannelActor, msg: ClientMessage): Unit =
66     msg match
67       case ClientMessage.SendMessage(message) =>
68         // Single-message send without context; not used in normal flow
69         sendMessage(channel, ServerMessage.ErrorMessage("Please use the chat interface to send messages"))
70 
71       case ClientMessage.EditMessage(message) =>
72         sendMessage(channel, ServerMessage.MessageUpdated(message))
73 
74       case ClientMessage.DeleteMessage(messageId) =>
75         sendMessage(channel, ServerMessage.MessageDeleted(messageId))
76 
77       case ClientMessage.RegenerateResponse(afterMessageId) =>
78         // Client will send the conversation history separately
79         sendMessage(channel, ServerMessage.MessageDeleted(afterMessageId))
80 
81       case ClientMessage.StopStreaming(messageId) =>
82         logger.info(s"[AIChat] Stop streaming requested for message $messageId")
83         activeStreams.get(channel).foreach(_.remove(messageId))
84 
85       case ClientMessage.ClearChat() =>
86         sendMessage(channel, ServerMessage.ChatCleared())
87 
88       case ClientMessage.ListModels(apiKey) =>
89         if apiKey.nonEmpty then
90           Future:
91             fetchModels(channel, apiKey)
92           .recover:
93             case ex: Exception =>
94               logger.error(s"[AIChat] Error fetching models: ${ex.getMessage}", ex)
95               sendMessage(channel, ServerMessage.ErrorMessage(s"Failed to fetch models: ${ex.getMessage}"))
96         else
97           sendMessage(channel, ServerMessage.ErrorMessage("Please set your API key first"))
98 
99       case ClientMessage.SpeakMessage(messageId, text, apiKey, voice, prompt) =>
100         if apiKey.nonEmpty then
101           Future:
102             generateTTS(channel, messageId, text, apiKey, voice, prompt)
103           .recover:
104             case ex: Exception =>
105               logger.error(s"[AIChat] TTS error: ${ex.getMessage}", ex)
106               sendMessage(channel, ServerMessage.TTSError(messageId, s"TTS failed: ${ex.getMessage}"))
107         else
108           sendMessage(channel, ServerMessage.TTSError(messageId, "Please set your API key first"))

110   private def generateResponse(channel: cask.WsChannelActor, apiKey: String, messages: Seq[ChatMessage], model: String): Unit =
111     // Create a new message ID for the response
112     val responseId = java.util.UUID.randomUUID().toString
113 
114     // Build the messages for OpenAI API
115     val openAIMessages = messages.map: msg =>
116       val contentParts = mutable.ArrayBuffer[ujson.Value]()
117 
118       // Add text content
119       if msg.content.nonEmpty then
120         contentParts += ujson.Obj("type" -> "text", "text" -> msg.content)
121 
122       // Add images if present
123       msg.images.foreach: imageData =>
124         val cleanBase64 = imageData.replaceFirst("^data:image/[^;]+;base64,", "")
125         contentParts += ujson.Obj(
126           "type" -> "image_url",
127           "image_url" -> ujson.Obj("url" -> s"data:image/png;base64,$cleanBase64")
128         )
129 
130       val role = msg.role match
131         case MessageRole.System => "system"
132         case MessageRole.User => "user"
133         case MessageRole.Assistant => "assistant"
134 
135       if contentParts.length == 1 && msg.images.isEmpty then
136         ujson.Obj("role" -> role, "content" -> msg.content)
137       else
138         ujson.Obj("role" -> role, "content" -> ujson.Arr(contentParts.toSeq*))
139 
140     // Make the API request with streaming
141     val url = "https://api.openai.com/v1/chat/completions"
142     val requestBody = ujson.Obj(
143       "model" -> model,
144       "messages" -> ujson.Arr(openAIMessages*),
145       "stream" -> true,
146       "stream_options" -> ujson.Obj("include_usage" -> true)
147     )
148 
149     // Send initial message to indicate streaming has started
150     val initialMessage = ChatMessage(responseId, MessageRole.Assistant, "")
151     sendMessage(channel, ServerMessage.MessageAdded(initialMessage))
152 
153     // Register this stream as active (for cancellation support)
154     activeStreams.getOrElseUpdate(channel, mutable.Set.empty).add(responseId)
155 
156     Future:
157       streamOpenAIRequest(channel, responseId, url, apiKey, requestBody, model)
158     .recover:
159       case ex: Exception =>
160         logger.error(s"[AIChat] Error generating response: ${ex.getMessage}", ex)
161         sendMessage(channel, ServerMessage.ErrorMessage(s"Error: ${ex.getMessage}"))

167   private def fetchModels(channel: cask.WsChannelActor, apiKey: String): Unit =
168     val request = java.net.http.HttpRequest.newBuilder()
169       .uri(java.net.URI.create("https://api.openai.com/v1/models"))
170       .header("Authorization", s"Bearer $apiKey")
171       .timeout(java.time.Duration.ofSeconds(30))
172       .GET()
173       .build()
174 
175     try
176       val response = httpClient.send(request, java.net.http.HttpResponse.BodyHandlers.ofString())
177       if response.statusCode() != 200 then
178         logger.error(s"[AIChat] OpenAI models API error: ${response.body()}")
179         sendMessage(channel, ServerMessage.ErrorMessage(s"Failed to fetch models (status ${response.statusCode()})"))
180       else
181         val json = ujson.read(response.body())
182         val allModels = json("data").arr.map(_("id").str).toSeq
183 
184         // Filter to chat-capable models (gpt-*, o1-*, o3-*, o4-*, chatgpt-*)
185         val chatModels = allModels
186           .filter: id =>
187             id.startsWith("gpt-") ||
188             id.startsWith("o1-") ||
189             id.startsWith("o3-") ||
190             id.startsWith("o4-") ||
191             id.startsWith("chatgpt-")
192           .filterNot: id =>
193             id.contains("instruct") ||
194             id.contains("realtime") ||
195             id.contains("audio") ||
196             id.contains("tts") ||
197             id.contains("whisper") ||
198             id.contains("transcribe") ||
199             id.contains("embedding")
200           .sorted
201 
202         logger.info(s"[AIChat] Fetched ${chatModels.size} chat models")
203         sendMessage(channel, ServerMessage.ModelsListed(chatModels))
204     catch
205       case ex: Exception =>
206         logger.error(s"[AIChat] Error fetching models: ${ex.getMessage}", ex)
207         sendMessage(channel, ServerMessage.ErrorMessage(s"Failed to fetch models: ${ex.getMessage}"))

212   private def streamOpenAIRequest(
213       channel: cask.WsChannelActor,
214       messageId: String,
215       url: String,
216       apiKey: String,
217       requestBody: ujson.Obj,
218       model: String
219   ): Unit =
220     val request = java.net.http.HttpRequest.newBuilder()
221       .uri(java.net.URI.create(url))
222       .header("Content-Type", "application/json")
223       .header("Authorization", s"Bearer $apiKey")
224       .header("Accept", "text/event-stream")
225       .timeout(java.time.Duration.ofSeconds(120))
226       .POST(java.net.http.HttpRequest.BodyPublishers.ofString(requestBody.toString))
227       .build()
228 
229     var promptTokens = 0
230     var completionTokens = 0
231 
232     try
233       val response = httpClient.send(request, java.net.http.HttpResponse.BodyHandlers.ofInputStream())
234 
235       if response.statusCode() != 200 then
236         val errorResponse = scala.io.Source.fromInputStream(response.body(), "UTF-8").mkString
237         response.body().close()
238         logger.error(s"[AIChat] OpenAI API error: $errorResponse")
239         sendMessage(channel, ServerMessage.ErrorMessage(s"API error (status ${response.statusCode()})"))
240         return
241 
242       val inputStream = response.body()
243       val reader = new java.io.BufferedReader(new java.io.InputStreamReader(inputStream, "UTF-8"), 1)
244 
245       var line: String = null
246       var cancelled = false
247       while !cancelled && { line = reader.readLine(); line != null } do
248         if !isStreamActive(channel, messageId) then
249           cancelled = true
250           logger.info(s"[AIChat] Stream $messageId cancelled by user")
251         else if line.startsWith("data: ") then
252           val data = line.substring(6)
253           if data != "[DONE]" then
254             Try(ujson.read(data)) match
255               case Success(json) =>
256                 // Extract usage from the final chunk (when stream_options include_usage is set)
257                 json.obj.get("usage").foreach: usage =>
258                   Try:
259                     promptTokens = usage("prompt_tokens").num.toInt
260                     completionTokens = usage("completion_tokens").num.toInt
261 
262                 // Extract content delta
263                 Try:
264                   val delta = json("choices")(0)("delta")
265                   if delta.obj.contains("content") then
266                     val chunk = delta("content").str
267                     sendMessage(channel, ServerMessage.StreamingChunk(messageId, chunk))
268               case Failure(_) => // Ignore parse errors for SSE
269 
270       reader.close()
271       inputStream.close()
272 
273       sendMessage(channel, ServerMessage.StreamingComplete(messageId, model, promptTokens, completionTokens))
274 
275     catch
276       case ex: Exception =>
277         logger.error(s"[AIChat] Streaming error: ${ex.getMessage}", ex)
278         sendMessage(channel, ServerMessage.ErrorMessage(s"Streaming error: ${ex.getMessage}"))
279     finally
280       activeStreams.get(channel).foreach(_.remove(messageId))

321       case ex: Exception =>
